package main

import (
	"bufio"
	"bytes"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"regexp"
	"strings"
)

type input struct {
	Pattern       string `json:"pattern"`
	Path          string `json:"path"`
	FileGlob      string `json:"file_glob"`
	CaseSensitive *bool  `json:"case_sensitive"`
}

type match struct {
	Path string `json:"path"`
	Line int    `json:"line"`
	Text string `json:"text"`
}

var ignoredDirs = map[string]bool{
	"node_modules": true, ".git": true, "venv": true, ".venv": true,
	"__pycache__": true, "dist": true, "build": true,
}

const maxMatches = 500

func resolve(rel string) (string, error) {
	if rel == "" {
		rel = "."
	}
	wd, err := os.Getwd()
	if err != nil {
		return "", err
	}
	// An absolute rel is used literally, not joined onto wd — Join
	// doesn't special-case an absolute second argument, it just
	// concatenates, so an absolute path already inside wd (e.g. wd itself
	// is the SSH login directory and rel is a full path under it) would
	// otherwise get silently doubled into wd+rel instead of resolving to
	// the real path the caller meant.
	var full string
	if filepath.IsAbs(rel) {
		full = filepath.Clean(rel)
	} else {
		full = filepath.Join(wd, rel)
	}
	r, err := filepath.Rel(wd, full)
	if err != nil || strings.HasPrefix(r, "..") {
		return "", fmt.Errorf("path escapes the working directory: %s", rel)
	}
	return full, nil
}

func isBinary(data []byte) bool {
	if len(data) > 8000 {
		data = data[:8000]
	}
	return bytes.IndexByte(data, 0) != -1
}

func emit(success bool, data any) {
	json.NewEncoder(os.Stdout).Encode(map[string]any{"success": success, "data": data})
}

func fail(msg string) { emit(false, msg) }

func main() {
	var in input
	if err := json.NewDecoder(os.Stdin).Decode(&in); err != nil {
		fail("invalid input: " + err.Error())
		return
	}
	if in.Pattern == "" {
		fail(`"pattern" is required`)
		return
	}

	root, err := os.Getwd()
	if err != nil {
		fail(err.Error())
		return
	}

	finalPattern := in.Pattern
	if in.CaseSensitive != nil && !*in.CaseSensitive {
		finalPattern = "(?i)" + in.Pattern
	}
	re, err := regexp.Compile(finalPattern)
	if err != nil {
		fail(fmt.Sprintf("invalid pattern %q: %v", in.Pattern, err))
		return
	}

	resolved, err := resolve(in.Path)
	if err != nil {
		fail(err.Error())
		return
	}
	info, err := os.Stat(resolved)
	if err != nil {
		if os.IsNotExist(err) {
			fail(fmt.Sprintf("path not found: %s", in.Path))
			return
		}
		fail(err.Error())
		return
	}

	var matches []match
	truncated := false

	searchOneFile := func(full string) error {
		if in.FileGlob != "" {
			ok, err := filepath.Match(in.FileGlob, filepath.Base(full))
			if err != nil || !ok {
				return nil
			}
		}
		f, err := os.Open(full)
		if err != nil {
			return nil
		}
		defer f.Close()

		sniff := make([]byte, 8000)
		n, _ := f.Read(sniff)
		if isBinary(sniff[:n]) {
			return nil
		}
		if _, err := f.Seek(0, 0); err != nil {
			return nil
		}

		rel, err := filepath.Rel(root, full)
		if err != nil {
			rel = full
		}

		scanner := bufio.NewScanner(f)
		scanner.Buffer(make([]byte, 0, 64*1024), 1024*1024)
		lineNum := 0
		for scanner.Scan() {
			lineNum++
			if len(matches) >= maxMatches {
				truncated = true
				return nil
			}
			line := scanner.Text()
			if re.MatchString(line) {
				if len(line) > 500 {
					line = line[:500] + "..."
				}
				matches = append(matches, match{Path: filepath.ToSlash(rel), Line: lineNum, Text: line})
			}
		}
		return nil
	}

	if !info.IsDir() {
		if err := searchOneFile(resolved); err != nil {
			fail(err.Error())
			return
		}
	} else {
		err = filepath.Walk(resolved, func(full string, fi os.FileInfo, err error) error {
			if err != nil {
				return err
			}
			if fi.IsDir() {
				if full != resolved && ignoredDirs[fi.Name()] {
					return filepath.SkipDir
				}
				return nil
			}
			if truncated {
				return filepath.SkipAll
			}
			return searchOneFile(full)
		})
		if err != nil {
			fail(fmt.Sprintf("failed to search %s: %v", in.Path, err))
			return
		}
	}

	emit(true, map[string]any{
		"pattern": in.Pattern, "matches": matches, "count": len(matches), "truncated": truncated,
	})
}
